# inside functions -- which is how scikit-image and scipy were missed the first
# time and crashed the worker at runtime:

#   metrics.max_diameter_mm   -> from skimage.measure import label, regionprops
#   density.solid_component   -> from scipy.ndimage import binary_erosion
#   medsam2.largest_component -> from skimage import measure
#
# tests/test_backend.py::test_worker_requirements_cover_src_imports now checks
# this list against every import in src/, so the next lazy import cannot slip
# through the same way.
#
# medpy is deliberately absent: it is only reached by metrics.hd95, which is an
# evaluation metric. measure() does not call it, so the worker never needs it.
